3.3. src/config/auth.js — JWT helper

const jwt = require('jsonwebtoken');
const bcrypt = require('bcryptjs');

// Parolu heşlə (bcrypt — geri qaytarılmaz)
async function hashPassword(parol) {
  return bcrypt.hash(parol, 10);
}

// Parolu yoxla
async function verifyPassword(parol, hash) {
  return bcrypt.compare(parol, hash);
}

// JWT tokeni yarat
function signToken(payload) {
  return jwt.sign(payload, process.env.JWT_SECRET,
                  { expiresIn: process.env.JWT_EXPIRES_IN || '7d' });
}

// JWT tokeni yoxla
function verifyToken(token) {
  return jwt.verify(token, process.env.JWT_SECRET);
}

module.exports = { hashPassword, verifyPassword, signToken, verifyToken };

3.4. src/middlewares/auth.js — qoruyucu

const { verifyToken } = require('../config/auth');

// İcazə tələb edən marşrutlara əlavə edilir
function authMiddleware(req, res, next) {
  const auth = req.headers.authorization || '';
  const token = auth.startsWith('Bearer ') ? auth.slice(7) : null;
  if (!token) return res.status(401).json({ xeta: 'Token tələb olunur' });
  try {
    req.user = verifyToken(token);   // { user_id, role, ... }
    next();
  } catch (e) {
    return res.status(401).json({ xeta: 'Yanlış və ya müddəti bitmiş token' });
  }
}

// Rol əsaslı məhdudiyyət
function roleMiddleware(...icazeler) {
  return (req, res, next) => {
    if (!icazeler.includes(req.user.role)) {
      return res.status(403).json({ xeta: 'İcazəniz yoxdur' });
    }
    next();
  };
}

module.exports = { authMiddleware, roleMiddleware };

4.1. src/models/layiheModel.js

const { pool } = require('../config/db');

// Bütün layihələr (view ilə tam icmal)
async function findAll() {
  const { rows } = await pool.query(
    `SELECT * FROM hesabat.layihe_tam_icmal ORDER BY layihe_id`
  );
  return rows;
}

// ID ilə bir layihə
async function findById(layiheId) {
  const { rows } = await pool.query(
    `SELECT * FROM layihe.layihe WHERE layihe_id = $1 AND silinib = FALSE`,
    [layiheId]
  );
  return rows[0];
}

// Yeni layihə (parametrlər $1, $2... — SQL injection qoruması)
async function create(data) {
  const { muessise_id, is_novu_id, seher_id, kod, ad, plan_budce,
          bashlama_tarixi, son_tarix } = data;
  const { rows } = await pool.query(
    `INSERT INTO layihe.layihe
       (muessise_id, is_novu_id, seher_id, kod, ad, plan_budce, bashlama_tarixi, son_tarix)
     VALUES ($1,$2,$3,$4,$5,$6,$7,$8)
     RETURNING *`,
    [muessise_id, is_novu_id, seher_id, kod, ad, plan_budce, bashlama_tarixi, son_tarix]
  );
  return rows[0];
}

// Plan büdcəni yenilə
async function updateBudget(layiheId, plan_budce) {
  const { rows } = await pool.query(
    `UPDATE layihe.layihe SET plan_budce = $2 WHERE layihe_id = $1 RETURNING *`,
    [layiheId, plan_budce]
  );
  return rows[0];
}

module.exports = { findAll, findById, create, updateBudget };

const layiheModel = require('../models/layiheModel');

// GET /api/layihe

5.4. Auth controller — login

const { hashPassword, verifyPassword, signToken } = require('../config/auth');
const istifadeciModel = require('../models/istifadeciModel');

async function login(req, res, next) {
  try {
    const { username, parol } = req.body;
    const istifadeci = await istifadeciModel.findByUsername(username);
    if (!istifadeci) return res.status(401).json({ xeta: 'İstifadəçi tapılmadı' });

    const duz = await verifyPassword(parol, istifadeci.password_hash);
    if (!duz) return res.status(401).json({ xeta: 'Şifrə yanlışdır' });

    // Giriş logu (audit.giris_log)
    await pool.query(
      `INSERT INTO audit.giris_log (istifadeci_id, istifadeci_ad, ip, success)
       VALUES ($1,$2,$3,TRUE)`,
      [istifadeci.user_id, istifadeci.username, req.ip]
    );

    const token = signToken({ user_id: istifadeci.user_id,
                              username: istifadeci.username,
                              role: istifadeci.role });
    res.json({ token, istifadeci: { user_id: istifadeci.user_id,
                                    username: istifadeci.username,
                                    role: istifadeci.role } });
  } catch (e) { next(e); }
}

module.exports = { login };

6 Express Tətbiqi və Server

6.1. src/app.js

const express = require('express');
const cors = require('cors');
const rateLimit = require('express-rate-limit');
const routes = require('./routes');
const { xetaMiddleware } = require('./middlewares/error');

const app = express();
app.use(express.json());

// CORS — frontend ünvanları
const originlar = (process.env.CORS_ORIGIN || '').split(',');
app.use(cors({ origin: originlar }));

// Sürət məhdudiyyəti
const limiter = rateLimit({
  windowMs: Number(process.env.RATE_LIMIT_WINDOW_MS) || 900000,
  max: Number(process.env.RATE_LIMIT_MAX) || 200,
});
app.use('/api', limiter);

// Health check — Dərsin əsas yoxlama nöqtəsi
app.get('/health', async (req, res) => {
  try {
    const vaxt = await require('./config/db').testConnection();
    res.json({ status: 'ok', baza: vaxt, vaxt: new Date().toISOString() });
  } catch (e) {
    res.status(503).json({ status: 'xeta', baza: e.message });
  }
});

app.use('/api', routes);

// 404 və xəta idarəçisi
app.use((req, res) => res.status(404).json({ xeta: 'Marşrut tapılmadı' }));
app.use(xetaMiddleware);

module.exports = app;

6.2. src/middlewares/error.js

function xetaMiddleware(err, req, res, next) {
  // unikal sütun xətası (23505)
  if (err.code === '23505') {
    return res.status(409).json({ xeta: 'Bu qeyd artıq mövcuddur', detay: err.detail });
  }
  // xarici açar xətası (23503)
  if (err.code === '23503') {
    return res.status(400).json({ xeta: 'İstinad edilən qeyd tapılmadı', detay: err.detail });
  }
  console.error(err);
  res.status(500).json({ xeta: 'Daxili server xətası' });
}
module.exports = { xetaMiddleware };

7 10 Test (Jest + Supertest)

Jest test çərçivəsidir, Supertest HTTP sorğularını simulyasiya edir. tests/integration/api.test.js:

const request = require('supertest');
const app = require('../../src/app');

describe('DeepSeek-5 ERP API', () => {

  // 1. Health check
  test('GET /health → status ok', async () => {
    const res = await request(app).get('/health');
    expect(res.status).toBe(200);
    expect(res.body.status).toBe('ok');
  });

  // 2. Auth tələb olunan marşrut token olmadan 401 qaytarmalıdır
  test('GET /api/layihe tokensiz → 401', async () => {
    const res = await request(app).get('/api/layihe');
    expect(res.status).toBe(401);
  });

  // 3. Login ilə token əldə et (sintetik istifadəçi)
  test('POST /api/auth/login düzgün parol ilə → token', async () => {
    const res = await request(app)
      .post('/api/auth/login')
      .send({ username: 'test_admin', parol: 'test123' });
    expect(res.status).toBe(200);
    expect(res.body.token).toBeDefined();
  });

  // 4. Layihələrin siyahısı token ilə
  test('GET /api/layihe token ilə → 5 layihə', async () => {
    const login = await request(app)
      .post('/api/auth/login')
      .send({ username: 'test_admin', parol: 'test123' });
    const res = await request(app)
      .get('/api/layihe')
      .set('Authorization', 'Bearer ' + login.body.token);
    expect(res.status).toBe(200);
    expect(Array.isArray(res.body)).toBe(true);
  });

  // 5. Yeni layihə yaradılması
  test('POST /api/layihe yeni layihə yaradır', async () => {
    const login = await request(app)
      .post('/api/auth/login')
      .send({ username: 'test_admin', parol: 'test123' });
    const res = await request(app)
      .post('/api/layihe')
      .set('Authorization', 'Bearer ' + login.body.token)
      .send({
        muessise_id: 1, is_novu_id: 1, seher_id: 1,
        kod: 'T-2026-TEST', ad: 'Test layihəsi', plan_budce: 10000
      });
    expect(res.status).toBe(201);
    expect(res.body.kod).toBe('T-2026-TEST');
  });

  // 6. Müqavilə ödəniş vəziyyəti
  test('GET /api/muqavile → borc siyahısı', async () => {
    const login = await request(app)
      .post('/api/auth/login')
      .send({ username: 'test_admin', parol: 'test123' });
    const res = await request(app)
      .get('/api/muqavile')
      .set('Authorization', 'Bearer ' + login.body.token);
    expect(res.status).toBe(200);
  });

  // 7. Tender siyahısı
  test('GET /api/tender → tenderlər', async () => {
    const login = await request(app)
      .post('/api/auth/login')
      .send({ username: 'test_admin', parol: 'test123' });
    const res = await request(app)
      .get('/api/tender')
      .set('Authorization', 'Bearer ' + login.body.token);
    expect(res.status).toBe(200);
  });

  // 8. İşçilər siyahısı
  test('GET /api/isci → işçilər', async () => {
    const login = await request(app)
      .post('/api/auth/login')
      .send({ username: 'test_admin', parol: 'test123' });
    const res = await request(app)
      .get('/api/isci')
      .set('Authorization', 'Bearer ' + login.body.token);
    expect(res.status).toBe(200);
    expect(res.body.length).toBeGreaterThanOrEqual(1);
  });

  // 9. Yanlış parol → 401
  test('POST /api/auth/login yanlış parol → 401', async () => {
    const res = await request(app)
      .post('/api/auth/login')
      .send({ username: 'test_admin', parol: 'yanlis' });
    expect(res.status).toBe(401);
  });

  // 10. Mövcud olmayan marşrut → 404
  test('GET /api/movcuddeyil → 404', async () => {
    const res = await request(app).get('/api/movcuddeyil');
    expect(res.status).toBe(404);
  });
});
