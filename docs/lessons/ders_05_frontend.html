3 API Xidməti və Auth Util

3.1. src/utils/auth.js — token idarəsi

// Tokeni localStorage-da saxlayırıq
const TOKEN_KEY = 'deepseek_token';

export function tokenSaxla(token) {
  localStorage.setItem(TOKEN_KEY, token);
}
export function tokenAl() {
  return localStorage.getItem(TOKEN_KEY);
}
export function tokenSil() {
  localStorage.removeItem(TOKEN_KEY);
}
export function daxilOlubmu() {
  return Boolean(tokenAl());
}

3.2. src/services/api.js — fetch sarmalayıcı

import { tokenAl } from '../utils/auth';

const BASE = 'http://localhost:5001/api';

async function istek(url, seçimler = {}) {
  const basliqlar = { 'Content-Type': 'application/json' };
  const token = tokenAl();
  if (token) basliqlar['Authorization'] = 'Bearer ' + token;

  const res = await fetch(BASE + url, {
    ...seçimler,
    headers: { ...basliqlar, ...seçimler.headers },
  });

  if (res.status === 401) {
    // Token bitibsə — login səhifəsinə yönləndir
    window.location.href = '/login';
    throw new Error('Sessiya bitib');
  }
  if (!res.ok) {
    const xeta = await res.json().catch(() => ({}));
    throw new Error(xeta.xeta || 'Sorğu xətası');
  }
  return res.json();
}

// Auth
export const login = (username, parol) =>
  istek('/auth/login', { method: 'POST', body: JSON.stringify({ username, parol }) });

// Layihələr
export const layiheleriGetir = () => istek('/layihe');
export const layiheYarat = (data) =>
  istek('/layihe', { method: 'POST', body: JSON.stringify(data) });

// Tenderlər
export const tenderleriGetir = () => istek('/tender');
export const muqavileleriGetir = () => istek('/muqavile');

// Dashboard üçün xülasə
export const xulasaGetir = async () => {
  const [layiheler, tenderler, muqavileler] = await Promise.all([
    layiheleriGetir(), tenderleriGetir(), muqavileleriGetir(),
  ]);
  return { layiheler, tenderler, muqavileler };
};

4 Login Səhifəsi

// src/pages/Login.jsx
import { useState } from 'react';
import { useNavigate } from 'react-router-dom';
import { login } from '../services/api';
import { tokenSaxla } from '../utils/auth';

export default function Login() {
  const [username, setUsername] = useState('');
  const [parol, setParol] = useState('');
  const [xeta, setXeta] = useState('');
  const [yuklenir, setYuklenir] = useState(false);
  const naviqasiya = useNavigate();

  async function tesdiqle(e) {
    e.preventDefault();
    setYuklenir(true); setXeta('');
    try {
      const cavab = await login(username, parol);
      tokenSaxla(cavab.token);
      naviqasiya('/');   // dashboard-a keç
    } catch (err) {
      setXeta(err.message);
    } finally {
      setYuklenir(false);
    }
  }

  return (
    <div style={{ maxWidth: 380, margin: '4rem auto', padding: '2rem',
                   border: '1px solid #dde3e8', borderRadius: 12 }}>
      <h2 style={{ color: '#0f3d5c', marginBottom: '1rem' }}>DeepSeek-5 ERP</h2>
      <p style={{ color: '#6b7280', marginBottom: '1.5rem' }}>Daxil olun</p>
      {xeta && <p style={{ color: '#b91c1c' }}>{xeta}</p>}
      <form onSubmit={tesdiqle}>
        <input
          placeholder="İstifadəçi adı"
          value={username}
          onChange={(e) => setUsername(e.target.value)}
          style={girisStili} />
        <input
          type="password"
          placeholder="Şifrə"
          value={parol}
          onChange={(e) => setParol(e.target.value)}
          style={girisStili} />
        <button disabled={yuklenir}
          style={{ width: '100%', background: '#0e9f6e', color: '#fff',
                   padding: '.7rem', border: 'none', borderRadius: 8,
                   fontWeight: 600, cursor: 'pointer' }}>
          {yuklenir ? 'Yüklənir...' : 'Daxil ol'}
        </button>
      </form>
    </div>
  );
}

const girisStili = {
  width: '100%', padding: '.7rem', marginBottom: '.8rem',
  border: '1px solid #cbd5e1', borderRadius: 8, fontSize: '1rem',
};

6 Layihələr Səhifəsi

Yeni layihə yaratma formalı siyahı. Backend-ə POST /api/layihe ilə göndərir:

// src/pages/Layiheler.jsx
import { useEffect, useState } from 'react';
import { layiheleriGetir, layiheYarat } from '../services/api';

export default function Layiheler() {
  const [layiheler, setLayiheler] = useState([]);
  const [form, setForm] = useState({ kod: '', ad: '', plan_budce: '' });
  const [mesaj, setMesaj] = useState('');

  useEffect(() => {
    layiheleriGetir().then(setLayiheler).catch((e) => setMesaj(e.message));
  }, []);

  async function elaveEt(e) {
    e.preventDefault();
    try {
      const yeni = await layiheYarat({ ...form, plan_budce: Number(form.plan_budce) });
      setLayiheler((m) => [yeni, ...m]);
      setForm({ kod: '', ad: '', plan_budce: '' });
      setMesaj('Layihə əlavə olundu');
    } catch (err) { setMesaj(err.message); }
  }

  return (
    <div>
      <h1 style={{ color: '#0f3d5c', marginBottom: '1rem' }}>Layihələr</h1>
      {mesaj && <p style={{ color: mesaj.includes('xəta') ? '#b91c1c' : '#0e9f6e' }}>{mesaj}</p>}

      <form onSubmit={elaveEt}
            style={{ display: 'flex', gap: '.5rem', marginBottom: '1.5rem' }}>
        <input placeholder="Kod (L-2026-006)"
               value={form.kod} onChange={(e) => setForm({ ...form, kod: e.target.value })}
               style={inp} />
        <input placeholder="Ad"
               value={form.ad} onChange={(e) => setForm({ ...form, ad: e.target.value })}
               style={{ ...inp, flex: 2 }} />
        <input placeholder="Büdcə"
               type="number" value={form.plan_budce}
               onChange={(e) => setForm({ ...form, plan_budce: e.target.value })}
               style={{ ...inp, width: 140 }} />
        <button style={btn}>Əlavə et</button>
      </form>

      <ul style={{ listStyle: 'none', padding: 0 }}>
        {layiheler.map((l) => (
          <li key={l.layihe_id}
              style={{ background: '#f8fafb', border: '1px solid #e0e5ea',
                       borderRadius: 8, padding: '1rem', marginBottom: '.5rem' }}>
            <strong>{l.kod}</strong> — {l.layihe_adi}
            <span style={{ float: 'right', color: '#0f3d5c', fontWeight: 600 }}>
              {Number(l.plan_budce).toLocaleString('az-AZ')} AZN
            </span>
          </li>
        ))}
      </ul>
    </div>
  );
}

const inp = { padding: '.6rem', border: '1px solid #cbd5e1', borderRadius: 8, fontSize: '.9rem' };
const btn = { background: '#0e9f6e', color: '#fff', border: 'none',
              padding: '.6rem 1.2rem', borderRadius: 8, cursor: 'pointer', fontWeight: 600 };

      </table>

  );
}

8.1. src/App.jsx

import { BrowserRouter, Routes, Route, Navigate } from 'react-router-dom';
import Login from './pages/Login';
import Dashboard from './pages/Dashboard';
import Layiheler from './pages/Layiheler';
import Tenderler from './pages/Tenderler';
import Navbar from './components/Navbar';
import ProtectedRoute from './components/ProtectedRoute';

export default function App() {
  return (
    <BrowserRouter>
      <Routes>
        <Route path="/login" element={<Login />} />
        <Route element={<ProtectedRoute />}>
          <Route path="/" element={<Navbar />}>
            <Route index element={<Dashboard />} />
            <Route path="layiheler" element={<Layiheler />} />
            <Route path="tenderler" element={<Tenderler />} />
          </Route>
        </Route>
        <Route path="*" element={<Navigate to="/" replace />} />
      </Routes>
    </BrowserRouter>
  );
}

import { Navigate, Outlet } from 'react-router-dom';
import { daxilOlubmu } from '../utils/auth';

// Giriş olunmayıbsa → /login
export default function ProtectedRoute() {
  return daxilOlubmu() ? <Outlet /> : <Navigate to="/login" replace />;
}

9 Gözlənilən Nəticə və Yoxlama

  1. npm run dev (frontend) və npm run dev (backend) işləyin;
  2. Brauzerdə http://localhost:5174 açın → Login səhifəsi;
  3. Daxil olun (test istifadəçisi) → Dashboard-da 4 xülasə kartı və plan-fakt cədvəli;

Gözlənilən problem: Login üçün test istifadəçisi lazımdır. Dərs 3-də yaratdığınız test_admin / test123 istifadəçisindən istifadə edin.
