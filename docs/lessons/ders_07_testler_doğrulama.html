1.1. Plan-fakt təhlili (#1)

SELECT l.layihe_id, l.kod, l.ad AS layihe,
       l.plan_budce,
       COALESCE(SUM(x.mebleg), 0) AS fakt_xerc,
       l.plan_budce - COALESCE(SUM(x.mebleg), 0) AS qaliq,
       ROUND(100.0 * COALESCE(SUM(x.mebleg), 0) / NULLIF(l.plan_budce, 0), 2) AS istifade_faizi
FROM layihe.layihe l
LEFT JOIN maliyye.xerc x ON x.layihe_id = l.layihe_id
WHERE l.silinib = FALSE
GROUP BY l.layihe_id
ORDER BY l.layihe_id;

Gözlənilən: L-2026-001 → fakt 68,700 / plan 850,000 → istifadə faizi ≈ 8.08.

1.2. Tender qənaəti (#2)

SELECT t.kod AS tender_kod, t.ad AS tender,
       t.qiymet_serhedi AS elan_mebleg,
       ti.sirket_ad AS qalib_sirket, ti.teklif_mebleg,
       t.qiymet_serhedi - ti.teklif_mebleg AS qenayet,
       ROUND(100.0 * (t.qiymet_serhedi - ti.teklif_mebleg) / NULLIF(t.qiymet_serhedi, 0), 2) AS qenayet_faizi
FROM satinalma.tender t
JOIN satinalma.tender_istirakci ti ON ti.tender_id = t.tender_id AND ti.qalib = TRUE
WHERE t.status_id = (SELECT status_id FROM satinalma.tender_status WHERE kod='qalib')
ORDER BY qenayet DESC;

Gözlənilən: T-2026-001 → 850,000 − 780,000 = 70,000 AZN qənaət (8.24%). T-2026-003 → 620,000 − 590,000 = 30,000 AZN (4.84%).

1.3. Regionlar üzrə layihə statistikası (#5)

SELECT r.ad AS region,
       COUNT(l.layihe_id) AS layihe_sayi,
       COALESCE(SUM(l.plan_budce), 0) AS plan_budce,
       COALESCE(SUM(x.mebleg), 0) AS fakt_xerc
FROM ref.region r
LEFT JOIN ref.seher s ON s.region_id = r.region_id
LEFT JOIN layihe.layihe l ON l.seher_id = s.seher_id AND l.silinib = FALSE
LEFT JOIN maliyye.xerc x ON x.layihe_id = l.layihe_id
GROUP BY r.region_id
ORDER BY plan_budce DESC;

Gözlənilən: Bakı regionu ən çox plan büdcəyə malikdir (fasad təmiri + dam təmiri layihələri orada).

1.4. Layihə üzrə işçi sayı və orta maaş (#3)

Kadr planlaması üçün: hər layihədə neçə işçi, orta maaş nə qədər.

SELECT l.kod, l.ad AS layihe,
       COUNT(DISTINCT t.isci_id) AS isci_sayi,
       ROUND(AVG(i.maas), 2) AS orta_maas,
       COUNT(DISTINCT t.teyinat_id) AS teyinat_sayi
FROM layihe.layihe l
LEFT JOIN kadr.layihe_isci_teyinat t ON t.layihe_id = l.layihe_id
LEFT JOIN kadr.isci i ON i.isci_id = t.isci_id
WHERE l.silinib = FALSE
GROUP BY l.layihe_id
ORDER BY isci_sayi DESC;

Gözlənilən: L-2026-001-də 2 işçi təyinatı, L-2026-004-də 1, L-2026-005-də 1; digərləri 0.

1.5. Ən çox istifadə olunan materiallar (#4)

Təchizat optimallaşdırması: ən çox dəyəri olan materiallar.

SELECT mn.ad AS material, mn.vahid,
       SUM(lm.miqdar) AS toplam_miqdar,
       SUM(lm.miqdar * lm.qiymet) AS toplam_deyer
FROM layihe.layihe_material lm
JOIN ref.material_novu mn ON mn.material_novu_id = lm.material_novu_id
GROUP BY mn.material_novu_id
ORDER BY toplam_deyer DESC;

Gözlənilən: Tikinti kərpici (145,000 × 0.45 = 65,250 AZN) ən yüksək ümumi dəyərə malikdir; sonra Portland sement (420 × 210 = 88,200 AZN).

1.6. Müqavilə ödəniş vəziyyəti (#6) — Borc idarəsi

SELECT m.nomre, m.podratci,
       m.mebleg AS muqavile_mebleg,
       COALESCE(SUM(o.mebleg), 0) AS odenen,
       m.mebleg - COALESCE(SUM(o.mebleg), 0) AS qaliq_borc,
       ROUND(100.0 * COALESCE(SUM(o.mebleg), 0) / NULLIF(m.mebleg, 0), 2) AS odenis_faizi
FROM satinalma.muqavile m
LEFT JOIN maliyye.odenis o ON o.muqavile_id = m.muqavile_id
WHERE m.aktif = TRUE
GROUP BY m.muqavile_id
ORDER BY qaliq_borc DESC;

Gözlənilən: MQ-2026-015 → məbləğ 780,000, ödənilən 500,000, borc 280,000 (64.10%). MQ-2026-008 → məbləğ 590,000, ödənilən 450,000, borc 140,000 (76.27%).

1.7. İş növləri üzrə orta layihə müddəti (#8)

SELECT isn.ad AS is_novu,
       COUNT(l.layihe_id) AS layihe_sayi,
       ROUND(AVG(l.son_tarix - l.bashlama_tarixi), 0) AS orta_gun,
       ROUND(AVG(l.son_tarix - l.bashlama_tarixi) / 30.0, 1) AS orta_ay

1.9. Yeni sxemlərin analitik sorğularının doğrulanması

database/04_analitik_sorgular.sql faylı ai/sened/risk/keyfiyyet/logistika/tehlike/audit sxemləri üçün əlavə analitik sorğular təqdim edir. Ən vacibləri:

-- Risk növləri üzrə paylanma (RISK-2)
SELECT risk_novu, COUNT(*) AS sayi,
       ROUND(AVG(derece), 1) AS orta_derece,
       COUNT(*) FILTER (WHERE status = 'aciq') AS aciq_sayi
FROM risk.risk
GROUP BY risk_novu ORDER BY sayi DESC;
-- Gözlənilən: maliyye=2, hava=2, texniki=1, tedaruk=1, kadr=1, tehlike=1, huquqi=1

-- Uğursuz giriş cəhdləri (AUDİT-2)
SELECT istifadeci_ad, ip,
       COUNT(*) FILTER (WHERE success = FALSE) AS ugursuz,
       COUNT(*) FILTER (WHERE success = TRUE) AS ugurlu
FROM audit.giris_log
GROUP BY istifadeci_ad, ip
ORDER BY ugursuz DESC;
-- Gözlənilən: ramin.eliyev 1 uğursuz, elcin.memmedov 1 uğursuz

1.10. Trigger və funksiyaların doğrulaması

Dərs 2-dəki triggerlərin real işlədiyini aşağıdakı ssenari ilə yoxlayın:

-- 1. updated_at triggeri: region-i yenilə, updated_at dəyişməlidir
SELECT updated_at FROM ref.region WHERE region_id = 1;
-- qeyd et, sonra:
UPDATE ref.region SET qeyd = 'test' WHERE region_id = 1;
SELECT updated_at FROM ref.region WHERE region_id = 1;
-- → updated_at dəyişdi ✓

-- 2. Audit triggeri: xərc yaz, audit_log-da qeyd olmalıdır
INSERT INTO maliyye.xerc (layihe_id, madde_id, mebleg, tarix, tesvir)
VALUES (1, (SELECT madde_id FROM maliyye.budce_madde WHERE kod='DIG'),
        500, CURRENT_DATE, 'Audit testi');
SELECT sxem, cedvel, emeliyyat, yeni_deyer ->> 'xerc_id' AS xerc_id
FROM audit.audit_log WHERE cedvel = 'xerc'
ORDER BY audit_id DESC LIMIT 1;
-- → INSERT əməliyyatı izlənildi ✓ (test xərcini sonra silin)
DELETE FROM maliyye.xerc WHERE tesvir = 'Audit testi';

-- 3. Sənəd versiya triggeri
UPDATE sened.sened SET ad = ad WHERE sened_id = 1;
SELECT versiya FROM sened.sened WHERE sened_id = 1;
-- → versiya 1-dən 2-yə artdı ✓

1.11. Bütün strukturun bir sorğu ilə yoxlanması

Bazanın tam "sağlamlıq hesabatı" — sxem, cədvəl, funksiya və trigger sayları bir yerdə:

SELECT 'sxem' AS kategoriya, COUNT(DISTINCT schemaname) AS sayi
FROM pg_tables WHERE schemaname IN
  ('ref','layihe','satinalma','maliyye','kadr','hesabat',
   'ai','sened','risk','keyfiyyet','logistika','tehlike','audit')
UNION ALL
SELECT 'cedvel', COUNT(*) FROM pg_tables WHERE schemaname IN
  ('ref','layihe','satinalma','maliyye','kadr','hesabat',
   'ai','sened','risk','keyfiyyet','logistika','tehlike','audit')
UNION ALL
SELECT 'funksiya', COUNT(*) FROM pg_proc p
  JOIN pg_namespace n ON n.oid = p.pronamespace
  WHERE n.nspname IN ('ref','layihe','satinalma','maliyye','kadr','hesabat',
                      'ai','sened','risk','keyfiyyet','logistika','tehlike','audit')
UNION ALL
SELECT 'trigger', COUNT(*) FROM pg_trigger t
  JOIN pg_class c ON c.oid = t.tgrelid
  JOIN pg_namespace n ON n.oid = c.relnamespace
  WHERE NOT t.tgisinternal AND n.nspname NOT IN ('pg_catalog','information_schema');

Gözlənilən: sxem=13, cedvel=48, funksiya=18, trigger=32.

3.2. Vahid test — tests/unit/auth.test.js

Parol heşi və token funksiyalarını heç bir xarici asılılıq olmadan test edir:

const { hashPassword, verifyPassword, signToken, verifyToken } = require('../../src/config/auth');

describe('Auth util', () => {
  test('parol heşlənir və yoxlanır', async () => {
    const hash = await hashPassword('sirr123');
    expect(hash).not.toBe('sirr123');          // düz mətn saxlanmır
    const duz = await verifyPassword('sirr123', hash);
    expect(duz).toBe(true);
  });

  test('yanlış parol qəbul edilmir', async () => {
    const hash = await hashPassword('dogru');
    const duz = await verifyPassword('yanlis', hash);
    expect(duz).toBe(false);
  });

  test('token imzalanır və doğrulanır', () => {
    const token = signToken({ user_id: 1, role: 'admin' });
    const payload = verifyToken(token);
    expect(payload.user_id).toBe(1);
    expect(payload.role).toBe('admin');
  });

  test('korlanmış token xəta verir', () => {
    expect(() => verifyToken('eyni.deyil.gecerli')).toThrow();
  });
});

const request = require('supertest');

3.4. Model vahid testi — tests/unit/muqavileModel.test.js

Model testləri üçün real baza əvəzinə mock pool istifadə olunur. Bu, testləri sürətləndirir və bazadan asılı etmir:

jest.mock('../../src/config/db', () => ({
  pool: { query: jest.fn() },
}));

const { pool } = require('../../src/config/db');
const muqavileModel = require('../../src/models/muqavileModel');

describe('muqavileModel', () => {
  beforeEach(() => { jest.clearAllMocks(); });

  test('findAll borc vəziyyətini view-dən gətirir', async () => {
    pool.query.mockResolvedValue({ rows: [{ muqavile_id: 1, qaliq_borc: '280000.00' }] });

    const nəticə = await muqavileModel.findAll();
    expect(pool.query).toHaveBeenCalledWith(
      expect.stringContaining('hesabat.muqavile_odenis_veziyyeti')
    );
    expect(nəticə[0].qaliq_borc).toBe('280000.00');
  });

  test('findById yalnız aktiv müqaviləni qaytarır', async () => {
    pool.query.mockResolvedValue({ rows: [{ muqavile_id: 2, aktif: true }] });
    const m = await muqavileModel.findById(2);
    expect(m.aktif).toBe(true);
    expect(pool.query.mock.calls[0][1]).toEqual([2]);
  });
});

jest.mock() modulu saxta (fake) obyektlə əvəz edir — beləliklə pool.query real baza yerinə nəzarət edilən mock funksiyaya çevrilir.
